Main thick line cv2.line(img, p1, p2, color, thickness) # Inner glow highlight cv2.line(img, p1, p2, (255, 255, 255), round(thickness / 2)) # Outer glow layers glow_thick = round(thickness * glow_intensity) cv2.line(img, p1, p2, tuple(c//2 for c in color), glow_thick) cv2.line(img, p1, p2, (255, 255, 255), round(glow_thick / 2)) def draw_glowing_finger(img, positions, line_color, glow_size=12): """Draw glowing fingertips matching ring color""" tips = ['thumb_tip', 'index_tip', 'midle_tip', 'ring_tip', 'pinky_tip'] for tip_key in tips: if tip_key in positions: cx, cy = positions[tip_key] # Slightly thinner glowing lines for cross-connections draw_glowing_line(img, p1, p2, line_color, thickness=3) def calculate_distance(p1, p2): return math.hypot(p2[0] - p1[0], p2[1] - p1[1]) def transparent_overlay(base_img, overlay_img, x, y, size=None): if overlay_img is None: return base_img if size is not None: overlay_img = cv2.resize(overlay_img, size) h, w = overlay_img.shape[:2] img_h, img_w = base_img.shape[:2] #
